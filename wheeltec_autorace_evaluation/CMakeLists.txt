cmake_minimum_required(VERSION 3.16)
project(wheeltec_autorace_evaluation)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(ignition-transport11 REQUIRED)
find_package(ignition-msgs8 REQUIRED)
add_executable(ground_truth_bridge src/ground_truth_bridge.cpp)
target_compile_features(ground_truth_bridge PRIVATE cxx_std_17)
ament_target_dependencies(ground_truth_bridge rclcpp geometry_msgs)
target_link_libraries(ground_truth_bridge ignition-transport11::core ignition-msgs8::ignition-msgs8)
install(TARGETS ground_truth_bridge DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/evaluate_tracking scripts/export_reference scripts/plot_tracking
  DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(FILES README_EVALUATION.md DESTINATION share/${PROJECT_NAME})
ament_package()
